## 2. 주요 핵심 구현 코드

### 2.1 웹 스크립트(Client JS) 원격 직접 검증 및 로컬 동기화 (`base.html`)
```javascript
// app/templates/base.html
async function verifyAndSyncRemoteSession() {
    const defaultPrefix = '{{ default_prefix }}';
    const authVerifyUrl = '{{ auth_verify_url if auth_verify_url else "https://holyseeds.thewayworks.net/auth/api/verify-session" }}';

    // 1. 웹 브라우저에서 직접 원격 Keycloak verify-session 호출 (credentials: 'include'로 원격 쿠키 자동 전송)
    try {
        const remoteResp = await fetch(authVerifyUrl, {
            method: 'GET',
            credentials: 'include',
            headers: { 'Accept': 'application/json' }
        });
        if (remoteResp.ok) {
            const remoteData = await remoteResp.json();
            const isMgr = Boolean(
                remoteData.valid && (
                    remoteData.is_manager === true
                    || remoteData.is_super === true
                    || remoteData.has_required_role === true
                    || remoteData.role_flag === '1'
                    || remoteData.role_flag === '2'
                    || (Array.isArray(remoteData.roles) && (remoteData.roles.includes('manager') || remoteData.roles.includes('super') || remoteData.roles.includes('admin')))
                )
            );
            if (isMgr) {
                // 2. 검증된 세션을 로컬 백엔드에 동기화하여 localhost용 쿠키 발급
                const syncResp = await fetch(`${defaultPrefix}/api/auth/sync-session`, {
                    method: 'POST',
                    credentials: 'include',
                    headers: { 'Content-Type': 'application/json', 'Accept': 'application/json' },
                    body: JSON.stringify(remoteData)
                });
                if (syncResp.ok) {
                    const syncData = await syncResp.json();
                    if (syncData.token) {
                        document.cookie = `auth_session=${encodeURIComponent(syncData.token)}; path=/; SameSite=Lax`;
                        document.cookie = `local_manager_token=${encodeURIComponent(syncData.token)}; path=/; SameSite=Lax`;
                    }
                }
                return { valid: true, is_manager: true, user: remoteData.user };
            }
        }
    } catch (remoteErr) {
        console.warn('[Remote Verify Error]', remoteErr);
    }
    return { valid: false, is_manager: false, user: null };
}
```

### 2.2 로컬 백엔드 세션 동기화 엔드포인트 (`admin_router.py`)
```python
# app/routers/admin_router.py
@router.post("/api/auth/sync-session")
async def sync_remote_session(request: Request):
    data = await request.json()
    is_valid = bool(data.get("valid") is True)
    roles = data.get("roles", [])
    role_flag = str(data.get("role_flag", "")).strip()

    is_manager = bool(
        data.get("is_manager") is True
        or data.get("is_super") is True
        or data.get("has_required_role") is True
        or role_flag in ("1", "2")
        or ("manager" in roles)
    )

    if not (is_valid and is_manager):
        return JSONResponse(status_code=400, content={"status": "error", "synced": False})

    user_info = data.get("user") or {}
    token = create_local_manager_token(user_data=user_info)

    response = JSONResponse(content={"status": "ok", "synced": True, "is_manager": True, "token": token})
    response.set_cookie(key="local_manager_token", value=token, path="/", samesite="lax", max_age=86400)
    response.set_cookie(key="auth_session", value=token, path="/", samesite="lax", max_age=86400)
    return response
```
