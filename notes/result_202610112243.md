### 2.2 비동기 세션 검증 및 무한 루프 차단 가드
```python
# app/core/auth.py
async def verify_admin_session(auth_session: str) -> bool:
    """Validate user credentials and check for manager status using Keycloak API Proxy."""
    if not auth_session:
        return False
    try:
        async with httpx.AsyncClient() as client:
            response = await client.get(
                AUTH_VERIFY_URL,
                params={"session_id": auth_session},
                timeout=4.0
            )
            if response.status_code == 200:
                result = response.json()
                return result.get("valid") is True and result.get("is_manager") is True
    except Exception as e:
        logger.warning("Failed to verify Keycloak session via %s: %s", AUTH_VERIFY_URL, e)
    return False

async def require_manager_web(request: Request) -> bool:
    user = await get_current_user_from_request(request)
    if user and user.get("is_manager"):
        return True

    # Loop Breaker: 이미 1회 리다이렉트 후 복귀했는데도 세션이 없으면 403 Forbidden 즉시 반환
    if request.query_params.get("auth_attempted") == "1":
        raise HTTPException(status_code=403, detail="관리자(manager) 권한이 확인되지 않았습니다.")

    full_target_url = str(request.url)
    redirect_target = full_target_url + ("&" if "?" in full_target_url else "?") + "auth_attempted=1"
    encoded_redirect = urllib.parse.quote(redirect_target, safe="")
    login_url = f"{AUTH_URL}/login?redirect={encoded_redirect}&require_role=manager"
    raise HTTPException(status_code=303, headers={"Location": login_url})
```

### 2.3 프론트엔드 사전 검증 및 분기 제어 (`handleManagerAction`)
```javascript
// app/templates/base.html
window.handleManagerAction = async function(event, targetPath, onSuccessCallback) {
  if (event && event.preventDefault) event.preventDefault();
  
  // 1. 브라우저 저장소 및 쿠키에서 세션 토큰 확인
  let sessionToken = getCookie('auth_session') || sessionStorage.getItem('auth_session') || localStorage.getItem('auth_session');
  
  // 2. 내부 엔드포인트를 통한 사전 검증 (verify_admin_session)
  let statusUrl = '/votes/api/auth/status';
  if (sessionToken) {
    statusUrl += '?session_id=' + encodeURIComponent(sessionToken);
  }

  try {
    const res = await fetch(statusUrl, { credentials: 'same-origin' });
    if (res.ok) {
      const data = await res.json();
      if (data.is_manager) {
        // 이미 매니저로 검증된 경우: 타겟 경로로 직접 즉시 이동 (원격 Keycloak 리다이렉트 차단)
        if (typeof onSuccessCallback === 'function') {
          onSuccessCallback();
        } else if (targetPath) {
          window.location.href = targetPath;
        }
        return false;
      }
    }
  } catch (err) {
    console.warn("Pre-verification failed:", err);
  }

  // 3. 미인증 또는 매니저 권한 부재 시: Keycloak 로그인 페이지로 리다이렉트 (origin 포함 복귀 URL 보장)
  const origin = window.location.origin;
  const fullTarget = (targetPath.startsWith('http://') || targetPath.startsWith('https://'))
    ? targetPath
    : (origin + (targetPath.startsWith('/') ? targetPath : '/' + targetPath));
  const loginUrl = 'https://holyseeds.thewayworks.net/auth/login?redirect=' + encodeURIComponent(fullTarget) + '&require_role=manager';
  window.location.href = loginUrl;
  return false;
};
```

- 원격 서버에 이미 로그인이 되어 있거나 세션 토큰을 가진 경우, 버튼 클릭 시 내부 시스템 사전 검증을 통해 Keycloak 리다이렉션 없이 즉시 관리자 페이지로 직행함을 확인하였습니다.
